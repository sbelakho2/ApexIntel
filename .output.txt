# Tangerine Standard Library — Serialization Framework
#
# Provides serialization and deserialization infrastructure with
# pluggable format support (JSON, TOML, YAML, MsgPack).
#
# Usage:
#   #[derive(Serialize, Deserialize)]
#   struct Person { name: String, age: Int }
#
#   let json = to_json(&person)
#   let person2 = from_json[Person](json)?
use std::core::{Option, Result, Vec, String, Bool, Int, Float, Hash, Eq}
use std::collections::{Map}
# ———————————————————————————————————————————————————————————————
# Serialize Trait
# ———————————————————————————————————————————————————————————————
trait Serialize
  def serialize(self: &Self, s: &mut Serializer) -> Result[Unit, SerError]
end
trait Deserialize
  def deserialize(d: &mut Deserializer) -> Result[Self, SerError]
end
# ———————————————————————————————————————————————————————————————
# Error Types
# ———————————————————————————————————————————————————————————————
enum SerError
  InvalidType(String)
  MissingField(String)
  UnexpectedToken(String)
  InvalidValue(String)
  Eof
  Io(String)
  Custom(String)
end
def ser_error_message(e: &SerError) -> String
  match e
  when SerError::InvalidType(s) then "invalid type: " + s.clone()
  when SerError::MissingField(s) then "missing field: " + s.clone()
  when SerError::UnexpectedToken(s) then "unexpected token: " + s.clone()
  when SerError::InvalidValue(s) then "invalid value: " + s.clone()
  when SerError::Eof then "unexpected end of input"
  when SerError::Io(s) then "IO error: " + s.clone()
  when SerError::Custom(s) then s.clone()
  end
end
impl Error for SerError
  def message(&self) -> String
    ser_error_message(self)
  end
end
# ———————————————————————————————————————————————————————————————
# Value Type (format-agnostic intermediate representation)
# ———————————————————————————————————————————————————————————————
enum Value
  Null
  Bool(Bool)
  Int(Int)
  Float(Float)
  String(String)
  Array(Vec[Value])
  Object(Map[String, Value])
end
def value_is_null(v: &Value) -> Bool
  match v
  when Value::Null then true
  when _ then false
  end
end
def value_as_bool(v: &Value) -> Option[Bool]
  match v
  when Value::Bool(b) then Option::Some(b)
  when _ then Option::None
  end
end
def value_as_int(v: &Value) -> Option[Int]
  match v
  when Value::Int(i) then Option::Some(i)
  when _ then Option::None
  end
end
def value_as_float(v: &Value) -> Option[Float]
  match v
  when Value::Float(f) then Option::Some(f)
  when Value::Int(i) then Option::Some(i as Float)
  when _ then Option::None
  end
end
def value_as_string(v: &Value) -> Option[String]
  match v
  when Value::String(s) then Option::Some(s.clone())
  when _ then Option::None
  end
end
def value_as_array(v: &Value) -> Option[&Vec[Value]]
  match v
  when Value::Array(arr) then Option::Some(&arr)
  when _ then Option::None
  end
end
def value_as_object(v: &Value) -> Option[&Map[String, Value]]
  match v
  when Value::Object(obj) then Option::Some(&obj)
  when _ then Option::None
  end
end
def value_get(v: &Value, key: &str) -> Option[&Value]
  match v
  when Value::Object(obj) then obj.get(key.to_string())
  when _ then Option::None
  end
end
def value_index(v: &Value, idx: UInt) -> Option[&Value]
  match v
  when Value::Array(arr) then arr.get(idx)
  when _ then Option::None
  end
end
# ———————————————————————————————————————————————————————————————
# Serializer Interface
# ———————————————————————————————————————————————————————————————
trait Serializer
  def serialize_bool(self: &mut Self, v: Bool) -> Result[Unit, SerError]
  def serialize_int(self: &mut Self, v: Int) -> Result[Unit, SerError]
  def serialize_float(self: &mut Self, v: Float) -> Result[Unit, SerError]
  def serialize_str(self: &mut Self, v: &str) -> Result[Unit, SerError]
  def serialize_bytes(self: &mut Self, v: &[u8]) -> Result[Unit, SerError]
  def serialize_none(self: &mut Self) -> Result[Unit, SerError]
  def serialize_some(self: &mut Self, v: &dyn Serialize) -> Result[Unit, SerError]
  def serialize_seq_begin(self: &mut Self, len: Option[UInt]) -> Result[Unit, SerError]
  def serialize_seq_element(self: &mut Self, v: &dyn Serialize) -> Result[Unit, SerError]
  def serialize_seq_end(self: &mut Self) -> Result[Unit, SerError]
  def serialize_map_begin(self: &mut Self, len: Option[UInt]) -> Result[Unit, SerError]
  def serialize_map_key(self: &mut Self, k: &str) -> Result[Unit, SerError]
  def serialize_map_value(self: &mut Self, v: &dyn Serialize) -> Result[Unit, SerError]
  def serialize_map_end(self: &mut Self) -> Result[Unit, SerError]
  def serialize_struct_begin(self: &mut Self, name: &str, len: UInt) -> Result[Unit, SerError]
  def serialize_struct_field(self: &mut Self, name: &str, v: &dyn Serialize) -> Result[Unit, SerError]
  def serialize_struct_end(self: &mut Self) -> Result[Unit, SerError]
end
# ———————————————————————————————————————————————————————————————
# Deserializer Interface
# ———————————————————————————————————————————————————————————————
trait Deserializer
  def deserialize_bool(self: &mut Self) -> Result[Bool, SerError]
  def deserialize_int(self: &mut Self) -> Result[Int, SerError]
  def deserialize_float(self: &mut Self) -> Result[Float, SerError]
  def deserialize_str(self: &mut Self) -> Result[String, SerError]
  def deserialize_bytes(self: &mut Self) -> Result[Vec[u8], SerError]
  def deserialize_option[T: Deserialize](self: &mut Self) -> Result[Option[T], SerError]
  def deserialize_seq[T: Deserialize](self: &mut Self) -> Result[Vec[T], SerError]
  def deserialize_map[K: Deserialize + Hash + Eq, V: Deserialize](self: &mut Self) -> Result[Map[K, V], SerError]
  def deserialize_struct[T: Deserialize](self: &mut Self, name: &str, fields: &[&str]) -> Result[T, SerError]
end
# ———————————————————————————————————————————————————————————————
# Built-in Serialize implementations
# ———————————————————————————————————————————————————————————————
impl Serialize for Bool
  def serialize(self: &Bool, s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_bool(self.clone())
  end
end
impl Serialize for Int
  def serialize(self: &Int, s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_int(self.clone())
  end
end
impl Serialize for Float
  def serialize(self: &Float, s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_float(self.clone())
  end
end
impl Serialize for String
  def serialize(self: &String, s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_str(self.as_str())
  end
end
impl[T: Serialize] Serialize for Option[T]
  def serialize(self: &Option[T], s: &mut Serializer) -> Result[Unit, SerError]
    match self
    when Option::Some(v) then s.serialize_some(&v)
    when Option::None then s.serialize_none()
    end
  end
end
impl[T: Serialize] Serialize for Vec[T]
  def serialize(self: &Vec[T], s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_seq_begin(Option::Some(self.len()))?
    for item in self do
      s.serialize_seq_element(&item)?
    end
    s.serialize_seq_end()
  end
end
impl[K: Serialize + Hash + Eq, V: Serialize] Serialize for Map[K, V]
  def serialize(self: &Map[K, V], s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_map_begin(Option::Some(self.len()))?
    for entry in self do
      let (k, v) = entry
      s.serialize_map_key(k.to_string().as_str())?
      s.serialize_map_value(&v)?
    end
    s.serialize_map_end()
  end
end
impl Serialize for Value
  def serialize(self: &Value, s: &mut Serializer) -> Result[Unit, SerError]
    match self
    when Value::Null then s.serialize_none()
    when Value::Bool(b) then s.serialize_bool(b.clone())
    when Value::Int(i) then s.serialize_int(i.clone())
    when Value::Float(f) then s.serialize_float(f.clone())
    when Value::String(str) then s.serialize_str(str.as_str())
    when Value::Array(arr) then arr.serialize(s)
    when Value::Object(obj) then
      s.serialize_map_begin(Option::Some(obj.len()))?
      for entry in obj do
        let (k, v) = entry
        s.serialize_map_key(k.as_str())?
        s.serialize_map_value(&v)?
      end
      s.serialize_map_end()
    end
  end
end

# ———————————————————————————————————————————————————————————————
# Derive Macro Support
# ———————————————————————————————————————————————————————————————
# The compiler recognizes #[derive(Serialize)] and #[derive(Deserialize)]
# and generates implementations based on struct field names and types.
#
# Example generated code for:
#   #[derive(Serialize, Deserialize)]
#   struct Point { x: Int, y: Int }
#
# becomes:
#   impl Serialize for Point
#     def serialize(self: &Point, s: &mut Serializer) -> Result[Unit, SerError]
#       s.serialize_struct_begin("Point", 2)?
#       s.serialize_struct_field("x", &self.x)?
#       s.serialize_struct_field("y", &self.y)?
#       s.serialize_struct_end()
#     end
#   end
#
#   impl Deserialize for Point
#     def deserialize(d: &mut Deserializer) -> Result[Point, SerError]
#       d.deserialize_struct[Point]("Point", &["x", "y"])
#     end
#   end
# ———————————————————————————————————————————————————————————————
# Field Attributes
# ———————————————————————————————————————————————————————————————
# Supported attributes for struct fields:
#
#   #[serde(rename = "other_name")]     — use a different name in serialized form
#   #[serde(skip)]                      — skip this field entirely
#   #[serde(skip_serializing)]          — skip only during serialization
#   #[serde(skip_deserializing)]        — skip only during deserialization
#   #[serde(default)]                   — use Default::default() if missing
#   #[serde(default = "my_default")]    — use specified function if missing
#   #[serde(flatten)]                   — flatten nested struct fields
#   #[serde(with = "module")]           — use custom serialize/deserialize
struct SerdeFieldAttr
  rename: Option[String]
  skip: Bool
  skip_serializing: Bool
  skip_deserializing: Bool
  use_default: Bool
  default_fn: Option[String]
  flatten: Bool
  with_module: Option[String]
end
def default_field_attr() -> SerdeFieldAttr
  SerdeFieldAttr {
    rename: Option::None,
    skip: false,
    skip_serializing: false,
    skip_deserializing: false,
    use_default: false,
    default_fn: Option::None,
    flatten: false,
    with_module: Option::None
  }
end

# Convenience Functions
# ———————————————————————————————————————————————————————————————
# Serialize to Value (format-agnostic)
def to_value[T: Serialize](v: &T) -> Result[Value, SerError]
  let mut s = ValueSerializer::new()
  v.serialize(&mut s)?
  Result::Ok(s.finish())
end
# Deserialize from Value
def from_value[T: Deserialize](v: &Value) -> Result[T, SerError]
  let mut d = ValueDeserializer::new(v)
  T::deserialize(&mut d)
end
# ———————————————————————————————————————————————————————————————
# ValueSerializer - serializes to Value type
# ———————————————————————————————————————————————————————————————
struct ValueSerializer
  stack: Vec[ValueBuilderState]
  result: Option[Value]
end
enum ValueBuilderState
  Root
  InArray(Vec[Value])
  InObject(Map[String, Value], Option[String])  # pending key
end
def ValueSerializer::new() -> ValueSerializer
  let mut stack = Vec::new()
  stack.push(ValueBuilderState::Root)
  ValueSerializer { stack: stack, result: Option::None }
end
impl ValueSerializer
  def finish(self: &ValueSerializer) -> Value
    self.result.clone().unwrap_or(Value::Null)
  end

  def push_value(self: &mut ValueSerializer, v: Value) -> Result[Unit, SerError]
    match self.stack.last_mut()
    when Option::Some(state) then
      match state
      when ValueBuilderState::Root then
        self.result = Option::Some(v)
      when ValueBuilderState::InArray(arr) then
        arr.push(v)
      when ValueBuilderState::InObject(obj, key_opt) then
        match key_opt.take()
        when Option::Some(key) then
          obj.insert(key, v)
        when Option::None then
          return Result::Err(SerError::Custom("value without key"))
        end
      end
      Result::Ok(())
    when Option::None then
      Result::Err(SerError::Custom("serializer stack empty"))
    end
  end
end
impl Serializer for ValueSerializer
  def serialize_bool(self: &mut ValueSerializer, v: Bool) -> Result[Unit, SerError]
    self.push_value(Value::Bool(v))
  end

  def serialize_int(self: &mut ValueSerializer, v: Int) -> Result[Unit, SerError]
    self.push_value(Value::Int(v))
  end

  def serialize_float(self: &mut ValueSerializer, v: Float) -> Result[Unit, SerError]
    self.push_value(Value::Float(v))
  end

  def serialize_str(self: &mut ValueSerializer, v: &str) -> Result[Unit, SerError]
    self.push_value(Value::String(v.to_string()))
  end

  def serialize_bytes(self: &mut ValueSerializer, v: &[u8]) -> Result[Unit, SerError]
    # Encode as array of ints
    let mut arr = Vec::new()
    for b in v do
      arr.push(Value::Int(b as Int))
    end
    self.push_value(Value::Array(arr))
  end

  def serialize_none(self: &mut ValueSerializer) -> Result[Unit, SerError]
    self.push_value(Value::Null)
  end

  def serialize_some(self: &mut ValueSerializer, v: &dyn Serialize) -> Result[Unit, SerError]
    v.serialize(self)
  end

  def serialize_seq_begin(self: &mut ValueSerializer, len: Option[UInt]) -> Result[Unit, SerError]
    self.stack.push(ValueBuilderState::InArray(Vec::new()))
    Result::Ok(())
  end

  def serialize_seq_element(self: &mut ValueSerializer, v: &dyn Serialize) -> Result[Unit, SerError]
    v.serialize(self)
  end

  def serialize_seq_end(self: &mut ValueSerializer) -> Result[Unit, SerError]
    match self.stack.pop()
    when Option::Some(ValueBuilderState::InArray(arr)) then
      self.push_value(Value::Array(arr))
    when _ then
      Result::Err(SerError::Custom("not in array"))
    end
  end

  def serialize_map_begin(self: &mut ValueSerializer, len: Option[UInt]) -> Result[Unit, SerError]
    self.stack.push(ValueBuilderState::InObject(Map::new(), Option::None))
    Result::Ok(())
  end

  def serialize_map_key(self: &mut ValueSerializer, k: &str) -> Result[Unit, SerError]
    match self.stack.last_mut()
    when Option::Some(ValueBuilderState::InObject(_, key_opt)) then
      *key_opt = Option::Some(k.to_string())
      Result::Ok(())
    when _ then
      Result::Err(SerError::Custom("not in object"))
    end
  end

  def serialize_map_value(self: &mut ValueSerializer, v: &dyn Serialize) -> Result[Unit, SerError]
    v.serialize(self)
  end

  def serialize_map_end(self: &mut ValueSerializer) -> Result[Unit, SerError]
    match self.stack.pop()
    when Option::Some(ValueBuilderState::InObject(obj, _)) then
      self.push_value(Value::Object(obj))
    when _ then
      Result::Err(SerError::Custom("not in object"))
    end
  end

  def serialize_struct_begin(self: &mut ValueSerializer, name: &str, len: UInt) -> Result[Unit, SerError]
    self.serialize_map_begin(Option::Some(len))
  end

  def serialize_struct_field(self: &mut ValueSerializer, name: &str, v: &dyn Serialize) -> Result[Unit, SerError]
    self.serialize_map_key(name)?
    self.serialize_map_value(v)
  end

  def serialize_struct_end(self: &mut ValueSerializer) -> Result[Unit, SerError]
    self.serialize_map_end()
  end
end
# ———————————————————————————————————————————————————————————————
# ValueDeserializer - deserializes from Value type
# ———————————————————————————————————————————————————————————————
struct ValueDeserializer
  value: &Value
  path: Vec[String]
end
def ValueDeserializer::new(v: &Value) -> ValueDeserializer
  ValueDeserializer { value: v, path: Vec::new() }
end
impl Deserializer for ValueDeserializer
  def deserialize_bool(self: &mut ValueDeserializer) -> Result[Bool, SerError]
    match self.value
    when Value::Bool(b) then Result::Ok(b.clone())
    when _ then Result::Err(SerError::InvalidType("expected bool"))
    end
  end

  def deserialize_int(self: &mut ValueDeserializer) -> Result[Int, SerError]
    match self.value
    when Value::Int(i) then Result::Ok(i.clone())
    when _ then Result::Err(SerError::InvalidType("expected int"))
    end
  end

  def deserialize_float(self: &mut ValueDeserializer) -> Result[Float, SerError]
    match self.value
    when Value::Float(f) then Result::Ok(f.clone())
    when Value::Int(i) then Result::Ok(i.clone() as Float)
    when _ then Result::Err(SerError::InvalidType("expected float"))
    end
  end

  def deserialize_str(self: &mut ValueDeserializer) -> Result[String, SerError]
    match self.value
    when Value::String(s) then Result::Ok(s.clone())
    when _ then Result::Err(SerError::InvalidType("expected string"))
    end
  end

  def deserialize_bytes(self: &mut ValueDeserializer) -> Result[Vec[u8], SerError]
    match self.value
    when Value::Array(arr) then
      let mut bytes = Vec::new()
      for v in arr do
        match v
        when Value::Int(i) then bytes.push(i.clone() as u8)
        when _ then return Result::Err(SerError::InvalidType("expected byte"))
        end
      end
      Result::Ok(bytes)
    when _ then Result::Err(SerError::InvalidType("expected array of bytes"))
    end
  end

  def deserialize_option[T: Deserialize](self: &mut ValueDeserializer) -> Result[Option[T], SerError]
    match self.value
    when Value::Null then Result::Ok(Option::None)
    when _ then
      let inner = T::deserialize(self)?
      Result::Ok(Option::Some(inner))
    end
  end

  def deserialize_seq[T: Deserialize](self: &mut ValueDeserializer) -> Result[Vec[T], SerError]
    match self.value
    when Value::Array(arr) then
      let mut result = Vec::new()
      for item in arr do
        let mut d = ValueDeserializer::new(&item)
        result.push(T::deserialize(&mut d)?)
      end
      Result::Ok(result)
    when _ then Result::Err(SerError::InvalidType("expected array"))
    end
  end

  def deserialize_map[K: Deserialize + Hash + Eq, V: Deserialize](self: &mut ValueDeserializer) -> Result[Map[K, V], SerError]
    match self.value
    when Value::Object(obj) then
      let mut result = Map::new()
      for entry in obj do
        let (k, v) = entry
        let mut kd = ValueDeserializer::new(&Value::String(k.clone()))
        let mut vd = ValueDeserializer::new(&v)
        result.insert(K::deserialize(&mut kd)?, V::deserialize(&mut vd)?)
      end
      Result::Ok(result)
    when _ then Result::Err(SerError::InvalidType("expected object"))
    end
  end

  def deserialize_struct[T: Deserialize](self: &mut ValueDeserializer, name: &str, fields: &[&str]) -> Result[T, SerError]
    # Struct deserialization is handled by the generated deserialize impl
    # which calls field-by-field deserialization
    T::deserialize(self)
  end
end
# ———————————————————————————————————————————————————————————————
# Schema Evolution / Versioned Serialization
# ———————————————————————————————————————————————————————————————
## Supports forward/backward compatible schema changes:
## - Adding optional fields (defaults when missing)
## - Removing fields (ignored during deserialization)
## - Renaming fields (aliasing)
trait VersionedSerialize: Serialize
  ## Return the schema version for this type
  def schema_version() -> UInt
  ## Serialize with version header
  def serialize_versioned(self: &Self, s: &mut Serializer) -> Result[Unit, SerError]
    s.serialize_struct_begin("__versioned", 2)?
    s.serialize_field("__v", &Self::schema_version())?
    s.serialize_field("data", self)?
    s.serialize_struct_end()
  end
end
trait VersionedDeserialize: Deserialize
  ## Deserialize with version migration
  def deserialize_versioned(d: &mut Deserializer) -> Result[Self, SerError]
  ## Migrate data from an older version to the current version
  def migrate(version: UInt, data: &Value) -> Result[Value, SerError]
end
## Default field value annotation support
struct FieldDefault[T]
  field_name: String
  default_value: T
end
## Schema descriptor for a type
struct Schema
  name: String
  version: UInt
  fields: Vec[SchemaField]
end
struct SchemaField
  name: String
  field_type: SchemaType
  required: Bool
  default_value: Option[Value]
  aliases: Vec[String]     ## Previous names for this field
  added_in_version: UInt
  removed_in_version: Option[UInt]
end
enum SchemaType
  Bool_
  Int_
  Float_
  String_
  Bytes_
  Array_(Box[SchemaType])
  Map_(Box[SchemaType], Box[SchemaType])
  Optional(Box[SchemaType])
  Struct_(String)  ## Reference by name
  Enum_(String)
end
impl Schema
  def new(name: &str, version: UInt) -> Schema
    Schema { name: name.to_string(), version: version, fields: Vec::new() }
  end
  def field(self: &mut Schema, name: &str, field_type: SchemaType, required: Bool) -> &mut Schema
    self.fields.push(SchemaField {
      name: name.to_string(),
      field_type: field_type,
      required: required,
      default_value: Option::None,
      aliases: Vec::new(),
      added_in_version: 1,
      removed_in_version: Option::None,
    })
    self
  end
  def field_with_default(self: &mut Schema, name: &str, field_type: SchemaType, default: Value) -> &mut Schema
    self.fields.push(SchemaField {
      name: name.to_string(),
      field_type: field_type,
      required: false,
      default_value: Option::Some(default),
      aliases: Vec::new(),
      added_in_version: 1,
      removed_in_version: Option::None,
    })
    self
  end
  ## Validate a Value against this schema
  def validate(self: &Schema, value: &Value) -> Result[Unit, Vec[String]]
    let mut errors = Vec::new()
    match value
    when Value::Object(obj) then
      for field in &self.fields do
        if field.required then
          let found = obj.contains_key(&field.name) ||
            field.aliases.iter().any(|a| obj.contains_key(a))
          if !found then
            errors.push("missing required field: ".to_string() + field.name.clone())
          end
        end
      end
    when _ then
      errors.push("expected object, got other type".to_string())
    end
    if errors.is_empty() then Result::Ok(()) else Result::Err(errors) end
  end
  ## Apply defaults to missing optional fields
  def apply_defaults(self: &Schema, value: &mut Value) -> Unit
    if let Value::Object(obj) = value then
      for field in &self.fields do
        if !obj.contains_key(&field.name) then
          if let Option::Some(default) = &field.default_value then
            obj.insert(field.name.clone(), default.clone())
          end
        end
        ## Resolve aliases
        for alias in &field.aliases do
          if let Option::Some(v) = obj.remove(alias) then
            if !obj.contains_key(&field.name) then
              obj.insert(field.name.clone(), v)
            end
          end
        end
      end
    end
  end
end

# Format Registry — Pluggable Serialization Formats
# ———————————————————————————————————————————————————————————————
## A format that can serialize/deserialize Values
trait Format
  def name(self: &Self) -> &str
  def serialize(self: &Self, value: &Value) -> Result[Vec[u8], SerError]
  def deserialize(self: &Self, data: &[u8]) -> Result[Value, SerError]
  def content_type(self: &Self) -> &str
end
## Registry of available formats
struct FormatRegistry
  formats: Map[String, Box[dyn Format]]
end
def format_registry_new() -> FormatRegistry
  FormatRegistry { formats: Map::new() }
end
impl FormatRegistry
  def register[F: Format + 'static](self: &mut FormatRegistry, format: F) -> Unit
    self.formats.insert(format.name().to_string(), Box::new(format))
  end
  def get(self: &FormatRegistry, name: &str) -> Option[&dyn Format]
    self.formats.get(&name.to_string()).map(|f| f.as_ref())
  end
  def serialize(self: &FormatRegistry, format_name: &str, value: &Value) -> Result[Vec[u8], SerError]
    match self.get(format_name)
    when Option::Some(fmt) then fmt.serialize(value)
    when Option::None then Result::Err(SerError::Custom("unknown format: ".to_string() + format_name.to_string()))
    end
  end
  def deserialize(self: &FormatRegistry, format_name: &str, data: &[u8]) -> Result[Value, SerError]
    match self.get(format_name)
    when Option::Some(fmt) then fmt.deserialize(data)
    when Option::None then Result::Err(SerError::Custom("unknown format: ".to_string() + format_name.to_string()))
    end
  end
  def available_formats(self: &FormatRegistry) -> Vec[String]
    self.formats.keys().cloned().collect()
  end
end
## Built-in JSON format hook
struct JsonFormat {}
impl Format for JsonFormat
  def name(self: &JsonFormat) -> &str = "json"
  def content_type(self: &JsonFormat) -> &str = "application/json"
  def serialize(self: &JsonFormat, value: &Value) -> Result[Vec[u8], SerError]
    let s = value_to_json_string(value)
    Result::Ok(s.as_bytes().to_vec())
  end
  def deserialize(self: &JsonFormat, data: &[u8]) -> Result[Value, SerError]
    let s = match String::from_utf8(data.to_vec())
    when Result::Ok(s) then s
    when Result::Err(_) then return Result::Err(SerError::InvalidValue("invalid UTF-8".to_string()))
    end
    json_string_to_value(&s)
  end
end

## Helper: create a default registry with all built-in formats
def default_format_registry() -> FormatRegistry
  let mut reg = format_registry_new()
  reg.register(JsonFormat {})
  reg.register(TomlFormat {})
  reg
end
## Convenience functions for simple Value <-> JSON string conversion
def value_to_json_string(value: &Value) -> String
  match value
  when Value::Null then "null".to_string()
  when Value::Bool(b) then if *b then "true" else "false" end.to_string()
  when Value::Int(i) then i.to_string()
  when Value::Float(f) then f.to_string()
  when Value::String(s) then "\"" + s.replace("\"", "\\\"") + "\""
  when Value::Array(arr) then
    let parts: Vec[String] = arr.iter().map(|v| value_to_json_string(v)).collect()
    "[" + parts.join(",") + "]"
  when Value::Object(obj) then
    let parts: Vec[String] = obj.iter().map(|(k, v)| {
      "\"" + k.replace("\"", "\\\"") + "\":" + value_to_json_string(v)
    }).collect()
    "{" + parts.join(",") + "}"
  end
end
def json_string_to_value(s: &str) -> Result[Value, SerError]
  ## Delegates to std::json parser
  Result::Err(SerError::Custom("use std::json for parsing".to_string()))
end
def value_to_toml_string(value: &Value) -> String
  ## Simplified TOML serialization
  match value
  when Value::Object(obj) then
    let mut parts = Vec::new()
    for (k, v) in obj do
      parts.push(k.clone() + " = " + value_to_toml_value(v))
    end
    parts.join("\n")
  when _ then "".to_string()
  end
end
def value_to_toml_value(value: &Value) -> String
  match value
  when Value::String(s) then "\"" + s.replace("\"", "\\\"") + "\""
  when Value::Int(i) then i.to_string()
  when Value::Float(f) then f.to_string()
  when Value::Bool(b) then if *b then "true" else "false" end.to_string()
  when _ then "\"\"".to_string()
  end
end
def toml_string_to_value(s: &str) -> Result[Value, SerError]
  ## Delegates to std::toml parser
  Result::Err(SerError::Custom("use std::toml for parsing".to_string()))
end

## Attributes that can be applied to fields for serde derive macros:
##   #[serde(rename = "...")]        — rename field in serialized form
##   #[serde(default)]               — use Default::default() if missing
##   #[serde(default = "func")]      — use func() if missing
##   #[serde(skip)]                  — skip serialization/deserialization
##   #[serde(skip_serializing)]      — skip during serialization only
##   #[serde(skip_deserializing)]    — skip during deserialization only
##   #[serde(flatten)]               — flatten nested struct
##   #[serde(alias = "...")]         — accept alternative name on deserialize
##   #[serde(with = "module")]       — use custom ser/de functions
##   #[serde(deny_unknown_fields)]   — error on unexpected fields
enum SerdeAttribute
  Rename(String)
  Default_
  DefaultWith(String)
  Skip
  SkipSerializer
  SkipDeserializer
  Flatten
  Alias(String)
  With(String)
  DenyUnknownFields
end
